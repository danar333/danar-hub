local flyObjects, flyConn = {}, nil

local flyInputConns = {}

local function cleanFly()
    if flyConn then flyConn:Disconnect() flyConn = nil end
    for _, conn in ipairs(flyInputConns) do conn:Disconnect() end
    flyInputConns = {}
    for _, obj in pairs(flyObjects) do pcall(function() obj:Destroy() end) end
    flyObjects = {}
end

local function applyFly(on)
    cleanFly()
    if not on then return end
    local char = player.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    local hum  = getHumanoid()
    if not root or not hum then return end

    local bv = Instance.new("BodyVelocity")
    bv.Velocity = Vector3.zero
    bv.MaxForce = Vector3.new(1e9, 1e9, 1e9)
    bv.Parent   = root

    local bg = Instance.new("BodyGyro")
    bg.MaxTorque = Vector3.new(1e9, 1e9, 1e9)
    bg.D         = 100
    bg.CFrame    = root.CFrame
    bg.Parent    = root

    flyObjects = { bv, bg }

    flyConn = RunService.Heartbeat:Connect(function(dt)
        if not bv.Parent or not bg.Parent then return end
        local char2 = player.Character
        local r = char2 and char2:FindFirstChild("HumanoidRootPart")
        if not r then return end
        local cam = workspace.CurrentCamera
        local cf  = cam.CFrame
        local spd = State.FlySpeed
        local hum2 = char2:FindFirstChildOfClass("Humanoid")
        local md = hum2 and hum2.MoveDirection or Vector3.zero

        local flatLook = Vector3.new(cf.LookVector.X, 0, cf.LookVector.Z)
        flatLook = flatLook.Magnitude > 0 and flatLook.Unit or Vector3.zero
        local flatRight = Vector3.new(cf.RightVector.X, 0, cf.RightVector.Z)
        flatRight = flatRight.Magnitude > 0 and flatRight.Unit or Vector3.zero

        local fwd    = md:Dot(flatLook)
        local strafe = md:Dot(flatRight)
        local move   = cf.LookVector * fwd + cf.RightVector * strafe

        if move.Magnitude > 0 then
            bv.Velocity = move.Unit * spd
        else
            bv.Velocity = Vector3.zero
        end
        bg.CFrame = cf - cf.Position + r.Position
    end)
end

local flingSteppedConn = nil
local function setFlingGhost(on)
    -- ponytail: trik FE Fling klasik — collision sendiri dimatikan biar kita
    -- nembus ke tubuh target tanpa pantulan solver (sumber "geter-geter")
    if flingSteppedConn then flingSteppedConn:Disconnect() flingSteppedConn = nil end
    if not on then return end
    flingSteppedConn = RunService.Stepped:Connect(function()
        local c = player.Character
        if not c then return end
        for _, p in ipairs(c:GetDescendants()) do
            if p:IsA("BasePart") then p.CanCollide = false end
        end
    end)
end

local function flingLoop()
    -- ponytail: resep JackMcJagger15 — collision sendiri off + BodyThrust
    -- konstan di HRP; velocity acak malah dilawan solver dan nggak ngefek
    local char   = player.Character
    local myRoot = char and char:FindFirstChild("HumanoidRootPart")
    if not myRoot then return end
    setFlingGhost(true)
    local thrust = Instance.new("BodyThrust")
    thrust.Force   = Vector3.new(State.FlingPower, 0, State.FlingPower)
    thrust.Location = Vector3.new(0, 0, -1)
    thrust.Parent   = myRoot
    while State.FlingOn do
        local c     = player.Character
        local r     = c and c:FindFirstChild("HumanoidRootPart")
        local tChar = State.FlingTarget and State.FlingTarget.Character
        local tRoot = tChar and tChar:FindFirstChild("HumanoidRootPart")
        if r and tRoot then
            if thrust.Parent ~= r then thrust.Parent = r end
            r.CFrame = tRoot.CFrame * CFrame.new(0, 0, -1)
        end
        task.wait()
    end
    pcall(function() thrust:Destroy() end)
    setFlingGhost(false)
end

local hitboxConn = nil
local hitboxOrig = {}
local function restoreHitboxes()
    for hrp, o in pairs(hitboxOrig) do
        if hrp.Parent then
            hrp.Size         = o.size
            hrp.Transparency = o.transparency
            hrp.Material     = o.material
            hrp.Massless     = o.massless
            hrp.CanCollide   = o.canCollide
        end
    end
    hitboxOrig = {}
end

local function setHitboxes(on)
    State.Hitbox = on
    if hitboxConn then hitboxConn:Disconnect() hitboxConn = nil end
    if on then
        hitboxConn = RunService.RenderStepped:Connect(function()
            for _, other in ipairs(Players:GetPlayers()) do
                if other ~= player then
                    local char2 = other.Character
                    local hrp = char2 and char2:FindFirstChild("HumanoidRootPart")
                    if hrp then
                        hitboxOrig[hrp] = hitboxOrig[hrp] or {
                            size = hrp.Size, transparency = hrp.Transparency,
                            material = hrp.Material, massless = hrp.Massless,
                            canCollide = hrp.CanCollide,
                        }
                        local s = State.HitboxSize
                        hrp.Size         = Vector3.new(s, s, s)
                        hrp.Transparency = 0.75
                        hrp.Massless     = true
                        hrp.CanCollide   = false
                    end
                end
            end
        end)
    else
        restoreHitboxes()
    end
end

local KORBLOX_MESH = {
    RightUpperLeg = { mesh = "rbxassetid://902942096", tex = "rbxassetid://902843398" },
    RightLowerLeg = { mesh = "rbxassetid://902942093", hide = true },
    RightFoot     = { mesh = "rbxassetid://902942089", hide = true },
}
local korbloxOriginals = {}
local cosmeticSpawnConn = nil

local function applyHeadlessChar(char)
    local head = char:FindFirstChild("Head") or char:WaitForChild("Head", 5)
    if not head then return end
    head.Transparency = 1
    for _, v in ipairs(head:GetChildren()) do
        if v:IsA("Decal") then v.Transparency = 1 end
    end
end

local function clearHeadlessChar(char)
    local head = char:FindFirstChild("Head")
    if not head then return end
    head.Transparency = 0
    for _, v in ipairs(head:GetChildren()) do
        if v:IsA("Decal") then v.Transparency = 0 end
    end
end

local function applyKorbloxChar(char)
    for name, cfg in pairs(KORBLOX_MESH) do
        local part = char:FindFirstChild(name) or char:WaitForChild(name, 5)
        if part then
            if not korbloxOriginals[part] then
                local okM, origMesh = pcall(function() return part.MeshId end)
                local okT, origTex = pcall(function() return part.TextureID end)
                korbloxOriginals[part] = {
                    mesh   = okM and origMesh or nil,
                    tex    = okT and origTex or nil,
                    transp = part.Transparency,
                }
            end
            pcall(function()
                part.MeshId = cfg.mesh
                if cfg.tex then part.TextureID = cfg.tex end
            end)
            if cfg.hide then part.Transparency = 1 end
        end
    end
end

local function clearKorbloxChar(char)
    for part, orig in pairs(korbloxOriginals) do
        if part.Parent then
            pcall(function()
                if orig.mesh then part.MeshId = orig.mesh end
                if orig.tex then part.TextureID = orig.tex end
            end)
            part.Transparency = orig.transp
        end
    end
    korbloxOriginals = {}
end

local function refreshCosmeticSpawn()
    if cosmeticSpawnConn then cosmeticSpawnConn:Disconnect() cosmeticSpawnConn = nil end
    if State.Headless or State.Korblox then
        cosmeticSpawnConn = player.CharacterAdded:Connect(function(c)
            task.wait(0.5)
            if State.Headless then applyHeadlessChar(c) end
            if State.Korblox then applyKorbloxChar(c) end
        end)
    end
end

local function setHeadless(on)
    State.Headless = on
    local char = player.Character
    if char then
        if on then applyHeadlessChar(char) else clearHeadlessChar(char) end
    end
    refreshCosmeticSpawn()
end

local function setKorblox(on)
    State.Korblox = on
    local char = player.Character
    if char then
        if on then applyKorbloxChar(char) else clearKorbloxChar(char) end
    end
    refreshCosmeticSpawn()
end

local ESP_NAME    = "danar_esp_highlight"
local ESP_BB_NAME = "danar_esp_bb"

local function attachESP(char)
    if not State.ESP then return end
    if char:FindFirstChild(ESP_NAME) then return end

    local root2 = char:FindFirstChild("HumanoidRootPart")
    local head2 = char:FindFirstChild("Head")
    if not root2 then return end

    local plrName = ""
    for _, p in ipairs(Players:GetPlayers()) do
        if p.Character == char then plrName = p.Name break end
    end

    local hl = Instance.new("Highlight")
    hl.Name             = ESP_NAME
    hl.FillColor        = THEME.Accent
    hl.FillTransparency = 0.65
    hl.OutlineColor     = Color3.new(1, 1, 1)
    hl.DepthMode        = Enum.HighlightDepthMode.AlwaysOnTop
    hl.Adornee          = char
    hl.Parent           = char

    local adornTarget = head2 or root2

    local bb = Instance.new("BillboardGui")
    bb.Name            = ESP_BB_NAME
    bb.AlwaysOnTop     = true
    bb.MaxDistance     = 600
    bb.Size            = UDim2.new(0, 120, 0, 22)
    bb.StudsOffsetWorldSpace = Vector3.new(0, 0, 0)
    bb.StudsOffset     = Vector3.new(0, 3.2, 0)
    bb.Adornee         = adornTarget
    bb.Parent          = char

    local nameLabel = Instance.new("TextLabel")
    nameLabel.BackgroundTransparency = 1
    nameLabel.Size            = UDim2.new(1, 0, 1, 0)
    nameLabel.Font            = Enum.Font.GothamBold
    nameLabel.TextSize        = 13
    nameLabel.TextColor3      = Color3.fromRGB(255, 255, 255)
    nameLabel.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
    nameLabel.TextStrokeTransparency = 0.4
    nameLabel.Text            = plrName
    nameLabel.TextScaled      = false
    nameLabel.Parent          = bb

    local distConn
    distConn = RunService.Heartbeat:Connect(function()
        if not bb.Parent or not State.ESP then
            distConn:Disconnect()
            return
        end
        local cam = workspace.CurrentCamera
        local myRoot = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
        if not myRoot or not adornTarget.Parent then return end

        local dist = (adornTarget.Position - myRoot.Position).Magnitude
        local scale = math.clamp(1 - (dist / 600) * 0.6, 0.4, 1)
        local sz    = math.floor(13 * scale + 0.5)
        if nameLabel.TextSize ~= sz then
            nameLabel.TextSize = sz
        end
        local alpha = math.clamp(1 - (dist / 600), 0.25, 1)
        nameLabel.TextTransparency = 1 - alpha
    end)
end

        if inst.Name == ESP_NAME or inst.Name == ESP_BB_NAME then
            inst:Destroy()
        end
    end
end

local stickyTarget = nil

local function measureTarget(p)
    local cam = workspace.CurrentCamera
    if not cam then return nil end
    local char = p.Character
    if not char then return nil end
    local head = char:FindFirstChild("Head")
    local hum  = char:FindFirstChildOfClass("Humanoid")
    if not head or not hum or hum.Health <= 0 then return nil end
    if State.AimTeamCheck and p.Team == player.Team then return nil end
    if State.AimWallCheck then
        local rparams = RaycastParams.new()
        rparams.FilterType = Enum.RaycastFilterType.Exclude
        rparams.FilterDescendantsInstances = { player.Character, char }
        local ray = workspace:Raycast(cam.CFrame.Position,
            head.Position - cam.CFrame.Position, rparams)
        if ray then return nil end
    end
    local screenPos, onScreen = cam:WorldToViewportPoint(head.Position)
    if not onScreen then return nil end
    local vpCenter = Vector2.new(cam.ViewportSize.X * 0.5, cam.ViewportSize.Y * 0.5)
    local dist2D = (Vector2.new(screenPos.X, screenPos.Y) - vpCenter).Magnitude
    if dist2D >= State.AimFOV then return nil end
    return dist2D
end

local function getNearestTarget()
    if stickyTarget and stickyTarget.Parent then
        if measureTarget(stickyTarget) then return stickyTarget end
    end
    stickyTarget = nil
    local cam = workspace.CurrentCamera
    local myRoot = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
    if not cam or not myRoot then return nil end
    local best, bestDist = nil, State.AimFOV
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= player then
            local d = measureTarget(p)
            if d and d < bestDist then
                bestDist = d
                best = p
            end
        end
    end
    stickyTarget = best
    return best
end

local fovCircle = nil

local function updateFOVCircle()
    if not fovCircle then return end
    local cam = workspace.CurrentCamera
    if not cam then return end
    local center = Vector2.new(cam.ViewportSize.X * 0.5, cam.ViewportSize.Y * 0.5)
    fovCircle.Position = center
    fovCircle.Radius   = State.AimFOV
    fovCircle.Color    = State.AimFOVColor
    fovCircle.Visible  = State.AimOn and not State.AimHideFOV
end

local function enableFOVCircle()

    fovCircle.Transparency = 1
    fovCircle.NumSides   = 64
    fovCircle.Radius     = State.AimFOV
    fovCircle.Visible    = true

end

bind(RunService.RenderStepped:Connect(function(dt)
    updateFOVCircle()
    if not State.AimOn then
        State._prevTargetPos = nil
        stickyTarget = nil
        if lockLabel then lockLabel.Visible = false end
        return
    end
    local target = getNearestTarget()
    if not target or not target.Character then
        State._prevTargetPos = nil
        if lockLabel then lockLabel.Visible = false end
        return
    end
    local aimPart = target.Character:FindFirstChild(State.AimPart or "Head")
        or target.Character:FindFirstChild("Head")
    if not aimPart then
        if lockLabel then lockLabel.Visible = false end
        return
    end
    lockLabel.Visible = true
    lockLabel.Text    = "Lock: " .. (target.DisplayName or target.Name)
    local aimPos = aimPart.Position
    if State.AimPrediction and State._prevTargetPos then
        local vel = (aimPart.Position - State._prevTargetPos) / dt
        aimPos = aimPart.Position + vel * State.AimPredStr
    end
    State._prevTargetPos = aimPart.Position
    local cam = workspace.CurrentCamera
    if State.AimSmooth <= 0 then
        cam.CFrame = CFrame.new(cam.CFrame.Position, aimPos)
    else
        local goalCF = CFrame.new(cam.CFrame.Position, aimPos)
        local alpha  = math.clamp(dt * (1 / math.max(State.AimSmooth, 0.01)), 0, 1)
        cam.CFrame   = cam.CFrame:Lerp(goalCF, alpha)
    end
end))

local function getMM2Role(plr)
    local char = plr.Character
    if not char then return "innocent" end
    for _, tool in ipairs(char:GetChildren()) do
        if tool:IsA("Tool") then
            local n = tool.Name:lower()
            if n:find("knife") or n:find("blade") or n:find("murder") then
                return "murderer"
            end
            if n:find("gun") or n:find("sheriff") or n:find("revolver") then
                return "sheriff"
            end
        end
    end
    local bp = plr:FindFirstChild("Backpack")
    if bp then
        for _, tool in ipairs(bp:GetChildren()) do
            if tool:IsA("Tool") then
                local n = tool.Name:lower()
                if n:find("knife") or n:find("blade") or n:find("murder") then
                    return "murderer"
                end
                if n:find("gun") or n:find("sheriff") or n:find("revolver") then
                    return "sheriff"
                end
            end
        end
    end
    return "innocent"
end

local function autoGrabGunLoop()
    while MM2.AutoGrabGun do
        local gun = findDroppedGun()
        if gun then
            local handle = gun:FindFirstChild("Handle")
            local char   = player.Character
            local root   = char and char:FindFirstChild("HumanoidRootPart")
            if handle and root then
                local dist = (handle.Position - root.Position).Magnitude
                if dist > 3 then
                    -- ponytail: durasi proporsional jarak, sama kayak coin farm
                    local sps = math.clamp(MM2.GrabSpeed, 1, 20) * 7
                    local tw = TweenService:Create(root,
                        TweenInfo.new(math.max(dist / sps, 0.08), Enum.EasingStyle.Linear),
                        { CFrame = CFrame.new(handle.Position + Vector3.new(0, 3, 0)) })
                    tw:Play()
                    tw.Completed:Wait()
                end
                task.wait(0.05)
                local hum = char and char:FindFirstChildOfClass("Humanoid")
                if hum then hum:EquipTool(gun) end
            end
        end
        task.wait(0.4)
    end
end

local function shootMurdererLoop()
    while MM2.ShootMurderer do
        local char = player.Character
        if char then
            local myTool = char:FindFirstChildOfClass("Tool")
            if myTool then
                local tn = myTool.Name:lower()
                local hasSheriffGun = tn:find("gun") or tn:find("sheriff") or tn:find("revolver")
                if hasSheriffGun then
                    for _, p in ipairs(Players:GetPlayers()) do
                        if p ~= player and getMM2Role(p) == "murderer" and p.Character then
                            local mHead = p.Character:FindFirstChild("Head")
                            local mRoot = p.Character:FindFirstChild("HumanoidRootPart")
                            local myRoot = char:FindFirstChild("HumanoidRootPart")
                            if mHead and mRoot and myRoot then
                                local dist = (mRoot.Position - myRoot.Position).Magnitude
                                if dist <= 200 then
                                    local remote = workspace:FindFirstChild("Shoot", true)
                                        or game:GetService("ReplicatedStorage"):FindFirstChild("Shoot", true)
                                        or game:GetService("ReplicatedStorage"):FindFirstChild("RemoteEvent", true)
                                    if remote and (remote:IsA("RemoteEvent") or remote:IsA("RemoteFunction")) then
                                        pcall(function()
                                            remote:FireServer(mHead.Position)
                                        end)
                                    else
                                        local cam = workspace.CurrentCamera
                                        if cam then
                                            cam.CFrame = CFrame.new(cam.CFrame.Position, mHead.Position)
                                        end
                                        if myTool:FindFirstChild("Fire") then
                                            pcall(function() myTool.Fire:FireServer(mHead.Position) end)
                                        end
                                    end
                                end
                            end
                        end
                    end
                end
            end
        end
        task.wait(0.35)
    end
end

local function bindOnce(name, fn)

    local gp = remotes and remotes:FindFirstChild("Gameplay")

    for _, v in ipairs(workspace:GetChildren()) do

            local cc = v:FindFirstChild("CoinContainer")

        end
    end

local function pcTp(cf)
    local char = player.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if root then root.CFrame = cf end
end

local function forceTouch(root, coin)
    if firetouchinterest then
        pcall(firetouchinterest, root, coin, 0)
        pcall(firetouchinterest, root, coin, 1)
    end
    local cd = coin:FindFirstChildOfClass("ClickDetector")
    if cd and fireclickdetector then pcall(fireclickdetector, cd) end
    local pp = coin:FindFirstChildOfClass("ProximityPrompt")
        or (coin.Parent and coin.Parent:FindFirstChildOfClass("ProximityPrompt"))
    if pp and fireproximityprompt then pcall(fireproximityprompt, pp) end
end

local function coinFarmLoop()
    ensureSafePart()
    local safeCF = safePart.CFrame + Vector3.new(0, 3, 0)
    while MM2.CoinFarm do
        local cc = roundActive and not bagFull and findCoinContainer()
        if cc then
            pcTp(safeCF)
            for _, coin in ipairs(cc:GetChildren()) do
                if not MM2.CoinFarm or bagFull then break end
                if coin:GetAttribute("CoinID") == MM2.CoinType
                   and coin:FindFirstChild("TouchInterest") then
                    for _ = 1, 7 do
                        if not MM2.CoinFarm then break end
                        pcTp(coin.CFrame + Vector3.new(0, 1, 0))
                        task.wait(0.03)
                    end
                    local char = player.Character
                    local root = char and char:FindFirstChild("HumanoidRootPart")
                    if root then forceTouch(root, coin) end
                    break -- satu coin per sweep, pola NoCapital
                end
            end
            pcTp(safeCF)
            task.wait(math.max(MM2.CoinDelay, 1) * (0.9 + math.random() * 0.2))
        else
            -- nunggu ronde / tas dipulihkan: parkir aman di bawah map
            if player.Character then pcTp(safeCF) end
            if bagFull and MM2.BagFullReset then
                local hum = player.Character and player.Character:FindFirstChildOfClass("Humanoid")
                if hum then hum.Health = 0 end
            end
            task.wait(0.5)
        end
    end
end

        if tabIcons[i] then tabIcons[i].TextColor3 = c end
        if tabChips[i] then tabChips[i].BackgroundTransparency = active and 0.82 or 0.93 end
        if tabPages[i] then tabPages[i].Visible = active end
    end
    indicator.Parent   = tabButtons[index]
    indicator.Position = UDim2.new(0, 2, 0.5, -10)
    indicator.Visible  = true
    local tw = TweenService:Create(indicator,

    indicator.Size = UDim2.fromOffset(3, 6)

end

    local function setFromX(x)
        local alpha = math.clamp((x - track.AbsolutePosition.X) / math.max(track.AbsoluteSize.X,1), 0, 1)
        local v = math.floor(min + (max-min)*alpha + 0.5)

            fill.Size     = UDim2.new(alpha, 0, 1, 0)
            knob.Position = UDim2.new(alpha, 0, 0.5, 0)

        or i.UserInputType == Enum.UserInputType.Touch) then setFromX(i.Position.X) end
    end))

        if i.UserInputType == Enum.UserInputType.MouseButton1
        or i.UserInputType == Enum.UserInputType.Touch then sliding = false end
    end))

local commFRemote
do
    local _commF = nil

        if _commF and _commF.Parent then return _commF end
        local rs = game:GetService("ReplicatedStorage")
        local remotes = rs:FindFirstChild("Remotes")
        _commF = remotes and remotes:FindFirstChild("CommF_")

    end

-- movement: smooth tween (proven pattern — no direct CFrame during travel)
local _farmTween = nil
local function atween(cf)
    local char = player.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not root then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if hum and hum.Sit then hum.Sit = false end
    if _farmTween then pcall(function() _farmTween:Cancel() end) _farmTween = nil end
    local dist = (cf.Position - root.Position).Magnitude
    if dist <= 10 then root.CFrame = cf; return end
    local speed = math.clamp(dist / 200, 0.1, 3)
    local tw = TweenService:Create(root,
        TweenInfo.new(speed, Enum.EasingStyle.Linear), { CFrame = cf })
    _farmTween = tw
    tw:Play()
end

local function btp(cf)
    local char = player.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if root then root.CFrame = cf end
end

-- FastAttack: paksa kecepatan animasi serangan ke 65x saat aktif (trik inti Alchemy)
do
    -- ponytail: timeout WAJIB — WaitForChild tanpa timeout bikin script
    -- menggantung selamanya kalau nama internal berubah pas game update
    local okRig, RigLib = pcall(function()
        local cf = game:GetService("ReplicatedStorage"):WaitForChild("CombatFramework", 9)
        return cf and cf:WaitForChild("RigLib", 9) and require(cf.RigLib)
    end)
    if okRig and type(RigLib) == "table" and RigLib.wrapAttackAnimationAsync then
        local oldWrap = RigLib.wrapAttackAnimationAsync
        RigLib.wrapAttackAnimationAsync = function(a, b, c, d, func)
            if BF.FastAtk then
                return oldWrap(a, b, c, 65, func)
            end
            return oldWrap(a, b, c, d, func)
        end
        BF._restoreWrap = function() RigLib.wrapAttackAnimationAsync = oldWrap end
    end
end

local function getController()
    local ps = player:FindFirstChild("PlayerScripts")
    local mod = ps and ps:FindFirstChild("CombatFramework")
    if not mod then return nil end
    local ok, cf = pcall(require, mod)
    if not ok or type(cf) ~= "table" then return nil end
    if getupvalue then
        local ok2, c = pcall(getupvalue, cf, 2)
        if ok2 and type(c) == "table" and c.attack then return c end
    end
    if type(cf.activeController) == "table" then return cf.activeController end
    return nil
end

local function equipAnyWeapon()
    local char = player.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    local bp = player:FindFirstChildOfClass("Backpack")
    if not hum or not bp then return end
    if char:FindFirstChildOfClass("Tool") then return end
    -- utamakan Melee/Sword; jangan sampe nyekap buah terus nyerang doang
    local best = nil
    for _, t in ipairs(bp:GetChildren()) do
        if t:IsA("Tool") then
            if t.ToolTip == "Melee" or t.ToolTip == "Sword" then
                best = t
                break
            end
            best = best or t
        end
    end
    if best then hum:EquipTool(best) end
end

local function nearestEnemy(monName)
    local enemies = workspace:FindFirstChild("Enemies")
    local char = player.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not enemies or not root then return nil end
    local best, bestD = nil, math.huge
    for _, e in ipairs(enemies:GetChildren()) do
        local eh = e:FindFirstChildOfClass("Humanoid")
        local ep = e:FindFirstChild("HumanoidRootPart")
        if eh and ep and eh.Health > 0 and (not monName or e.Name == monName) then
            local d = (ep.Position - root.Position).Magnitude
            if d < bestD then best, bestD = e, d end
        end
    end
    return best
end

local function questBracket()
    local list = QUESTS_ALL[BF_WORLD]
    if not list then return nil end
    local data = player:FindFirstChild("Data")
    local lvl = data and data:FindFirstChild("Level") and data.Level.Value or 0
    for _, b in ipairs(list) do
        if lvl >= b.min and lvl <= b.max then return b end
    end
    return nil
end

-- lock-on: tween ke offset + click spam, repeat-until musuh tumbang/mati
-- tanpa hitbox expansion (unreliable), offset kecil (5 stud di atas)
local function fightEnemy(v, quest)
    local vh = v:FindFirstChildOfClass("Humanoid")
    if not vh then return end
    local vp = v:FindFirstChild("HumanoidRootPart")
    if not vp then return end
    repeat
        task.wait()
        local c2 = player.Character
        local h2 = c2 and c2:FindFirstChildOfClass("Humanoid")
        if not (h2 and h2.Health > 0) then break end
        local rp = c2 and c2:FindFirstChild("HumanoidRootPart")
        if not rp then break end
        vp = v:FindFirstChild("HumanoidRootPart")
        if not vp then break end
        equipWeaponName(BF.SelectWeapon)
        autoHaki()
        BF.PosMon = vp.CFrame
        BF.NameMon = BF.NameMon or v.Name
        BF.StartMagnet = true
        BF.AttackOn = true
        -- tween smooth ke offset di atas mob (no jitter)
        atween(vp.CFrame * BF.Pos)
        -- disable collision musuh biar gak nge-block
        pcall(function()
            vp.CanCollide = false
            vh.WalkSpeed = 0
            local vhead = v:FindFirstChild("Head")
            if vhead then vhead.CanCollide = false end
        end)
        -- click spam
        pcall(function()
            local vu = game:GetService("VirtualUser")
            vu:CaptureController()
            vu:Button1Down(Vector2.new(1280, 672))
            vu:Button1Up(Vector2.new(1280, 672))
        end)
    until not BF.FarmOn
        or not v.Parent
        or not v:FindFirstChild("HumanoidRootPart")
        or not v:FindFirstChildOfClass("Humanoid")
        or v:FindFirstChildOfClass("Humanoid").Health <= 0
        or (quest and not questGuiVisible())
    BF.StartMagnet = false
    BF.AttackOn = false
end

-- ===== Main Farm Loop: Quest / NoQuest / Terdekat =====
local function bfFarmLoop()
    while BF.FarmOn do
        pcall(function()
            local char = player.Character
            local hum = char and char:FindFirstChildOfClass("Humanoid")
            local root = char and char:FindFirstChild("HumanoidRootPart")
            if not (hum and root and hum.Health > 0) then return end

            local r = commFRemote()

            if BF.FarmMode == "Quest" then
                -- Step 1: tentukan target dari level
                local b = questBracket()
                if not b then task.wait(0.5) return end
                BF.NameMon = b.mon

                -- Step 2: cek judul quest — kalau beda, buang
                local title = nil
                pcall(function()
                    local pg = player:FindFirstChild("PlayerGui")
                    title = pg and pg.Main and pg.Main.Quest
                        and pg.Main.Quest.Container
                        and pg.Main.Quest.Container.QuestTitle
                        and pg.Main.Quest.Container.QuestTitle.Title.Text
                end)
                if title and not title:find(b.mon) then
                    BF.StartMagnet = false
                    if r then pcall(function() r:InvokeServer("AbandonQuest") end) end
                end

                -- Step 3: quest belum ambil → pergi ke NPC + ambil
                if not questGuiVisible() then
                    BF.StartMagnet = false
                    local cfq = CFrame.new(b.x, b.y, b.z)
                    if BF.BypassTP
                       and (root.Position - cfq.Position).Magnitude > 1500 then
                        btp(cfq)
                    else
                        atween(cfq)
                    end
                    if (root.Position - cfq.Position).Magnitude <= 15 then
                        task.wait(0.15)
                        if r then
                            pcall(function() r:InvokeServer("StartQuest", b.q, b.lvl) end)
                        end
                    end
                    return
                end

                -- Step 4: quest aktif → cari + serang musuh
                local enemies = workspace:FindFirstChild("Enemies")
                if enemies and enemies:FindFirstChild(b.mon) then
                    for _, v in ipairs(enemies:GetChildren()) do
                        if not BF.FarmOn then break end
                        if v.Name == b.mon then
                            local vh = v:FindFirstChildOfClass("Humanoid")
                            local vp = v:FindFirstChild("HumanoidRootPart")
                            if vh and vp and vh.Health > 0 then
                                -- validasi judul lagi sebelum fight
                                local t2 = nil
                                pcall(function()
                                    local pg = player:FindFirstChild("PlayerGui")
                                    t2 = pg and pg.Main and pg.Main.Quest
                                        and pg.Main.Quest.Container
                                        and pg.Main.Quest.Container.QuestTitle
                                        and pg.Main.Quest.Container.QuestTitle.Title.Text
                                end)
                                if t2 and t2:find(b.mon) then
                                    fightEnemy(v, true)
                                else
                                    BF.StartMagnet = false
                                    if r then
                                        pcall(function() r:InvokeServer("AbandonQuest") end)
                                    end
                                end
                            end
                        end
                    end
                else
                    -- musuh habis → ke titik spawn + kejar preview respawn
                    BF.StartMagnet = false
                    BF.AttackOn = false
                    atween(CFrame.new(b.x, b.y, b.z))
                    unequipTool()
                    task.wait(0.2)
                    local preview = game:GetService("ReplicatedStorage"):FindFirstChild(b.mon)
                    local ph = preview and preview:FindFirstChild("HumanoidRootPart")
                    if ph then atween(ph.CFrame * CFrame.new(0, 10, 0)) end
                end

            elseif BF.FarmMode == "NoQuest" then
                -- ambil quest tanpa validasi judul
                local b = questBracket()
                if not b then task.wait(0.5) return end
                BF.NameMon = b.mon

                if not questGuiVisible() then
                    BF.StartMagnet = false
                    local cfq = CFrame.new(b.x, b.y, b.z)
                    if BF.BypassTP
                       and (root.Position - cfq.Position).Magnitude > 1500 then
                        btp(cfq)
                    else
                        atween(cfq)
                    end
                    if (root.Position - cfq.Position).Magnitude <= 15 then
                        task.wait(0.15)
                        if r then
                            pcall(function() r:InvokeServer("StartQuest", b.q, b.lvl) end)
                        end
                    end
                    return
                end

                local enemies = workspace:FindFirstChild("Enemies")
                if enemies and enemies:FindFirstChild(b.mon) then
                    for _, v in ipairs(enemies:GetChildren()) do
                        if not BF.FarmOn then break end
                        if v.Name == b.mon then
                            local vh = v:FindFirstChildOfClass("Humanoid")
                            local vp = v:FindFirstChild("HumanoidRootPart")
                            if vh and vp and vh.Health > 0 then
                                fightEnemy(v, false)
                            end
                        end
                    end
                else
                    BF.StartMagnet = false
                    BF.AttackOn = false
                    atween(CFrame.new(b.x, b.y, b.z))
                    unequipTool()
                    task.wait(0.2)
                    local preview = game:GetService("ReplicatedStorage"):FindFirstChild(b.mon)
                    local ph = preview and preview:FindFirstChild("HumanoidRootPart")
                    if ph then atween(ph.CFrame * CFrame.new(0, 10, 0)) end
                end

            else
                -- Terdekat: musuh terdekat berdasarkan jarak, tanpa quest
                local tgt = nearestEnemy(nil)
                if not tgt then
                    BF.StartMagnet = false
                    BF.AttackOn = false
                    task.wait(0.4)
                    return
                end
                BF.NameMon = tgt.Name
                fightEnemy(tgt, false)
            end
        end)
        task.wait()
    end
    BF.StartMagnet = false
    BF.AttackOn = false
    -- cleanup noclip body
    local char = player.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    local bv = root and root:FindFirstChild("BF_Noclip")
    if bv then bv:Destroy() end
end    BF.AttackOn = false
end

-- Rantai melee: Combat -> Black Leg -> Electro -> Fishman Karate -> Dragon Claw -> Superhuman
local function meleeChainStep()
    local data = player:FindFirstChild("Data")
    local bp = player:FindFirstChildOfClass("Backpack")
    local char = player.Character
    if not (data and bp and char) then return end
    local beli = data:FindFirstChild("Beli") and data.Beli.Value or 0
    local frag = data:FindFirstChild("Fragments") and data.Fragments.Value or 0
    local r = commFRemote()
    if not r then return end
    local has = function(n) return bp:FindFirstChild(n) or char:FindFirstChild(n) end
    local mastery = function(n)
        local t = bp:FindFirstChild(n) or char:FindFirstChild(n)
        return (t and t:FindFirstChild("Level")) and t.Level.Value or -1
    end
    pcall(function()
        if has("Combat") and beli >= 150000 then
            r:InvokeServer("BuyBlackLeg")
        elseif has("Black Leg") and mastery("Black Leg") >= 300 and beli >= 300000 then
            r:InvokeServer("BuyElectro")
        elseif has("Electro") and mastery("Electro") >= 300 and beli >= 750000 then
            r:InvokeServer("BuyFishmanKarate")
        elseif has("Fishman Karate") and mastery("Fishman Karate") >= 300 and frag >= 1500 then
            r:InvokeServer("BlackbeardReward", "DragonClaw", "1")
            r:InvokeServer("BlackbeardReward", "DragonClaw", "2")
        elseif has("Dragon Claw") and mastery("Dragon Claw") >= 300 and beli >= 3000000 then
            r:InvokeServer("BuySuperhuman")
        end
    end)
end

-- Helper: check tool existence across backpack+character
local function hasTool(name)
    local bp = player:FindFirstChild("Backpack")
    local ch = player.Character
    return (bp and bp:FindFirstChild(name)) or (ch and ch:FindFirstChild(name))
end
local function toolLevel(name)
    local t = hasTool(name)
    return t and t:FindFirstChild("Level") and t.Level.Value or 0
end
local function buyInvoke(method)
    local r = commFRemote()
    if r then pcall(function() r:InvokeServer(method) end) end
end
local function buyCheck(method)
    local r = commFRemote()
    if r then
        local ok, res = pcall(function() return r:InvokeServer(method, true) end)
        return ok and res == 1
    end
    return false
end
local function beliValue()
    local d = player:FindFirstChild("Data")
    return d and d:FindFirstChild("Beli") and d.Beli.Value or 0
end
local function fragValue()
    local d = player:FindFirstChild("Data")
    return d and d:FindFirstChild("Fragments") and d.Fragments.Value or 0
end

-- Auto Superhuman: Combat → Black Leg → Electro → Fishman → Dragon Claw → Superhuman
task.spawn(function()
    while task.wait(2) do
        if BF.AutoSuperhuman then
            pcall(function()
                -- kickstart: buy Combat if missing entirely
                if not hasTool("Combat") and beliValue() >= 150000 then
                    buyInvoke("BuyBlackLeg")
                end
                -- final: equip Superhuman
                if hasTool("Superhuman") then
                    BF.SelectWeapon = "Superhuman"
                    return
                end
                -- chain steps
                local chain = {
                    { tool = "Black Leg",     lvl = 300, beli = 300000,  buy = "BuyElectro" },
                    { tool = "Electro",       lvl = 300, beli = 750000,  buy = "BuyFishmanKarate" },
                    { tool = "Fishman Karate",lvl = 300, frag = 1500,    buy = "BlackbeardReward", arg1 = "DragonClaw", arg2 = "1" },
                    { tool = "Dragon Claw",   lvl = 300, beli = 3000000, buy = "BuySuperhuman" },
                }
                for _, step in ipairs(chain) do
                    local lv = toolLevel(step.tool)
                    if hasTool(step.tool) then
                        if lv >= step.lvl then
                            local cost = step.frag and fragValue() or beliValue()
                            local need = step.frag or step.beli
                            if cost >= need then
                                unequipTool()
                                task.wait(0.2)
                                if step.arg1 then
                                    local r = commFRemote()
                                    if r then
                                        pcall(function() r:InvokeServer(step.buy, step.arg1, "1") end)
                                        task.wait(0.3)
                                        pcall(function() r:InvokeServer(step.buy, step.arg1, "2") end)
                                    end
                                else
                                    buyInvoke(step.buy)
                                end
                            end
                        else
                            BF.SelectWeapon = step.tool
                        end
                        return
                    end
                end
                -- not holding anything in chain: buy Black Leg to start
                buyInvoke("BuyBlackLeg")
            end)
        end
    end
end)

safeTab("ESP", function()
    local pg = tabPages[7]
    sectionLabel(pg, "Visual")
    createToggle(pg, "ESP Pemain", false, function(v)
        State.ESP = v
        if v then
            local function hook(plr)
                if plr == player then return end
                if plr.Character then attachESP(plr.Character) end
                espCharConns[plr] = plr.CharacterAdded:Connect(function(c)
                    task.wait(0.3)
                    if State.ESP then attachESP(c) end
                end)
            end
            for _, plr in ipairs(Players:GetPlayers()) do hook(plr) end
            espPlayerConn = Players.PlayerAdded:Connect(hook)
            espRemoveConn = Players.PlayerRemoving:Connect(function(plr)
                if espCharConns[plr] then
                    espCharConns[plr]:Disconnect()
                    espCharConns[plr] = nil
                end
            end)
        else
            if espPlayerConn then espPlayerConn:Disconnect() espPlayerConn = nil end
            if espRemoveConn then espRemoveConn:Disconnect() espRemoveConn = nil end
            for _, conn in pairs(espCharConns) do conn:Disconnect() end
            espCharConns = {}
            clearESP()
        end
    end)
    createToggle(pg, "Fullbright", false, function(v)
        State.Fullbright = v setFullbright(v)
    end)
    createSlider(pg, "Field of View", 70, 120, 70, function(v)
        State.FOV = v setFOV(v)
    end)
end)

local uiOpenState = true

local function setOpen(open)
    if animBusy or open == uiOpenState or not screenGui.Parent then return end
    uiOpenState = open
    animBusy    = true
    if open then
        local ptw = TweenService:Create(pill,
            TweenInfo.new(0.12, Enum.EasingStyle.Quad, Enum.EasingDirection.In),
            { Size = UDim2.fromOffset(PILL_W, 0) })
        ptw.Completed:Connect(function() pill.Visible = false end)
        ptw:Play()
        main.Visible    = true
        main.AnchorPoint = Vector2.new(0.5, 0.5)
        main.Position   = savedPos or UDim2.new(0.5, 0, 0.5, 0)
        main.Size       = UDim2.fromOffset(MAIN_W, 0)
        local tw = TweenService:Create(main,
            TweenInfo.new(0.3, Enum.EasingStyle.Back, Enum.EasingDirection.Out),
            { Size = UDim2.fromOffset(MAIN_W, MAIN_H) })
        tw.Completed:Connect(function() animBusy = false end)
        tw:Play()
    else
        pill.Visible = true
        pill.Size    = UDim2.fromOffset(PILL_W, 0)
        TweenService:Create(pill,
            TweenInfo.new(0.24, Enum.EasingStyle.Back, Enum.EasingDirection.Out),
            { Size = UDim2.fromOffset(PILL_W, PILL_H) }):Play()
        local tw = TweenService:Create(main,
            TweenInfo.new(0.18, Enum.EasingStyle.Quad, Enum.EasingDirection.In),
            { Size = UDim2.fromOffset(MAIN_W, 0) })
        tw.Completed:Connect(function()
            main.Visible = false
            main.Size    = UDim2.fromOffset(MAIN_W, MAIN_H)
            animBusy     = false
        end)
        tw:Play()
    end
end

    if input.KeyCode == Enum.KeyCode.RightShift then setOpen(not uiOpenState) end
end))

local pillDragging  = false
local pillDragStart = nil
local pillStartPos  = nil

pillSymbol.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
       or input.UserInputType == Enum.UserInputType.Touch then
        pillDragging  = true
        pillDragStart = input.Position
        pillStartPos  = pill.Position
    end
end)

bind(UserInputService.InputChanged:Connect(function(input)
    if not pillDragging then return end
    if input.UserInputType ~= Enum.UserInputType.MouseMovement
       and input.UserInputType ~= Enum.UserInputType.Touch then return end
    local vp   = workspace.CurrentCamera.ViewportSize
    local limX = math.max(vp.X / 2 - PILL_W / 2, 0)
    local rawX = pillStartPos.X.Offset + (input.Position.X - pillDragStart.X)
    local rawY = math.clamp(
        pillStartPos.Y.Offset + (input.Position.Y - pillDragStart.Y),
        8, vp.Y - PILL_H - 8)
    pill.Position = UDim2.new(0.5, math.clamp(rawX, -limX, limX), 0, rawY)
end))

bind(UserInputService.InputEnded:Connect(function(input)
    if pillDragging and (input.UserInputType == Enum.UserInputType.MouseButton1
                         or input.UserInputType == Enum.UserInputType.Touch) then
        pillDragging = false
    end
end))

pillBtn.Activated:Connect(function()
    setOpen(not uiOpenState)
end)

local function runCleanup()
    State.FlingOn = false
    if BF then
        BF.FarmOn = false
        BF.AttackOn = false
        BF.FastAtk = false
        if BF._restoreWrap then pcall(BF._restoreWrap) end
    end
    do
        local c  = player.Character
        local r  = c and c:FindFirstChild("HumanoidRootPart")
        local th = r and r:FindFirstChildOfClass("BodyThrust")
        if th then pcall(function() th:Destroy() end) end
        if flingSteppedConn then flingSteppedConn:Disconnect() flingSteppedConn = nil end
    end
    State.Noclip = false
    State.InfiniteJump = false
    State.ESP = false
    if State.Fly then State.Fly = false applyFly(false) end
    if State.TPWalk then setTPWalk(false) end
    if State.AntiFling then setAntiFling(false) end
    if MM2.AimLockMurder then setAimLockMurder(false) end
    if State.Headless then setHeadless(false) end
    if State.Korblox then setKorblox(false) end
    if State.Fullbright then State.Fullbright = false setFullbright(false) end
    if State.FPSBoost then State.FPSBoost = false setFPSBoost(false) end
    if State.Hitbox then State.Hitbox = false setHitboxes(false) end
    clearESP()
    applySpeed()
    applyJump()
    if getgenv().DANAR_SET_ANTIKICK then getgenv().DANAR_SET_ANTIKICK(false) end
    State.AimOn = false
    disableFOVCircle()
    MM2.RoleESP       = false
    if roleESPJoinConn then
        roleESPJoinConn:Disconnect()
        roleESPJoinConn = nil
    end
    for _, conn in pairs(roleESPConns) do conn:Disconnect() end
    roleESPConns = {}
    MM2.AutoGrabGun   = false
    MM2.GunESP        = false
    clearGunESP()
    if gunTrackerLine then
        gunTrackerLine:Remove()
        gunTrackerLine = nil
    end
    MM2.ShootMurderer = false
    MM2.CoinFarm      = false
    pcall(function() RunService:Set3dRenderingEnabled(true) end)
    if safePart then pcall(function() safePart:Destroy() end) safePart = nil end
    clearRoleESP()
    clearGunESP()
    for _, conn in pairs(playerHooks) do pcall(function() conn:Disconnect() end) end
    table.clear(playerHooks)
    if espPlayerConn then espPlayerConn:Disconnect() espPlayerConn = nil end
    if espRemoveConn then espRemoveConn:Disconnect() espRemoveConn = nil end
    for _, conn in pairs(espCharConns) do pcall(function() conn:Disconnect() end) end
    table.clear(espCharConns)
    for _, conn in ipairs(connections) do
        pcall(function() conn:Disconnect() end)
    end
    table.clear(connections)
end

screenGui.Destroying:Connect(runCleanup)

applySpeed()
applyJump()
