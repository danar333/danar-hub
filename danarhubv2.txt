local currentTheme = "Midnight Purple"
local function applyTheme(name)
    local t = THEMES[name]
    if not t then return end
    local old = {}
    for k,v in pairs(THEME) do old[k]=v end
    for k,v in pairs(t) do THEME[k]=v end
    currentTheme = name
    local function dist(a,b) local dr=a.R-b.R; local dg=a.G-b.G; local db=a.B-b.B; return math.sqrt(dr*dr+dg*dg+db*db) end
    local function closest(oldCol)
        local bestK, bestD = nil, 1e9
        for k,c in pairs(old) do
            local d = dist(c, oldCol)
            if d < bestD then bestD=d; bestK=k end
        end
        return bestK, bestD
    end
    pcall(function()
        local gui
        -- 1. CoreGui
        pcall(function() gui = game:GetService("CoreGui"):FindFirstChild("DanarHubGui") end)
        -- 2. PlayerGui
        if not gui or not gui.Parent then pcall(function() gui = player:FindFirstChild("PlayerGui") and player.PlayerGui:FindFirstChild("DanarHubGui") end) end
        -- 3. gethui (executor hidden)
        if (not gui or not gui.Parent) and gethui then pcall(function() local h=gethui(); if h then gui = h:FindFirstChild("DanarHubGui") end end) end
        -- 4. fallback scan all
        if not gui or not gui.Parent then
            pcall(function()
                for _,v in ipairs(game:GetService("CoreGui"):GetChildren()) do if v.Name=="DanarHubGui" then gui=v break end end
            end)
        end
        if not gui or not gui.Parent then
            -- last try: cari lewat screenGui global jika ada
            pcall(function() if _G.DanarHubGui then gui=_G.DanarHubGui end end)
        end
        if not gui or not gui.Parent then return end
        for _, inst in ipairs(gui:GetDescendants()) do
            -- pill toggle khusus: size 40x22 + punya knob
            if inst:IsA("TextButton") and inst.Size == UDim2.fromOffset(40,22) then
                local knob = inst:FindFirstChildOfClass("Frame")
                local isOn = knob and knob.Position.X.Scale > 0.5
                inst.BackgroundColor3 = isOn and THEME.Accent or THEME.Off
            elseif inst:IsA("Frame") and inst:FindFirstChildOfClass("Frame") and inst.Size.Y.Offset==5 then
                -- slider track
                inst.BackgroundColor3 = THEME.Track
                for _, ch in ipairs(inst:GetChildren()) do
                    if ch:IsA("Frame") and ch.BackgroundColor3 then
                        -- fill
                        local k,d = closest(ch.BackgroundColor3)
                        if k=="Accent" or d<0.15 then ch.BackgroundColor3 = THEME.Accent end
                    end
                end
            else
                if inst:IsA("GuiObject") then
                    -- skip putih knob/slider
                    if not (inst.BackgroundColor3.R>0.98 and inst.BackgroundColor3.G>0.98 and inst.BackgroundColor3.B>0.98) then
                        local k,d = closest(inst.BackgroundColor3)
                        if k and (d < 0.12 or k=="Accent" or k=="Accent2") then inst.BackgroundColor3 = THEME[k] end
                    end
                end
                if (inst:IsA("TextLabel") or inst:IsA("TextButton")) then
                    -- jangan ubah putih knob (1,1,1)
                    if not (inst.TextColor3.R>0.98 and inst.TextColor3.G>0.98 and inst.TextColor3.B>0.98) then
                        local k,d = closest(inst.TextColor3)
                        if k and (d < 0.12 or k=="Accent" or k=="Accent2") then inst.TextColor3 = THEME[k] end
                    end
                end
                if inst:IsA("UIStroke") then
                    local k,d = closest(inst.Color)
                    if k and (d < 0.22 or k=="Accent" or k=="Accent2") then inst.Color = THEME[k] end
                end
                if inst:IsA("ScrollingFrame") then
                    inst.ScrollBarImageColor3 = THEME.Accent
                end
                if inst:IsA("TextLabel") and inst.Parent and inst.Parent:IsA("Frame") and inst.Text:find("FPS") then
                    -- status labels keep SubText
                end
            end
        end
        -- paksa container utama
        pcall(function()
            for _, o in ipairs(gui:GetDescendants()) do
                if o.Name=="Main" then o.BackgroundColor3 = THEME.Bg end
                if o.Name=="TitleBar" then o.BackgroundColor3 = THEME.Card end
                if o.Name=="DanarPill" then o.BackgroundColor3 = THEME.Card end
            end
        end)
        if State and State._fovCircle then
            pcall(function() State._fovCircle.Color = THEME.Accent end)
            State.AimFOVColor = THEME.Accent
        end
    end)
end

local flyObjects, flyConn = {}, nil

local flyInputConns = {}

local function cleanFly()
    if flyConn then flyConn:Disconnect() flyConn = nil end
    for _, conn in ipairs(flyInputConns) do conn:Disconnect() end
    flyInputConns = {}
    for _, obj in pairs(flyObjects) do pcall(function() obj:Destroy() end) end
    flyObjects = {}
end

local function applyFly(on)
    cleanFly()
    if not on then return end
    local char = player.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    local hum  = getHumanoid()
    if not root or not hum then return end

    local bv = Instance.new("BodyVelocity")
    bv.Velocity = Vector3.zero
    bv.MaxForce = Vector3.new(1e9, 1e9, 1e9)
    bv.Parent   = root

    local bg = Instance.new("BodyGyro")
    bg.MaxTorque = Vector3.new(1e9, 1e9, 1e9)
    bg.D         = 100
    bg.CFrame    = root.CFrame
    bg.Parent    = root

    flyObjects = { bv, bg }

    flyConn = RunService.Heartbeat:Connect(function(dt)
        if not bv.Parent or not bg.Parent then return end
        local char2 = player.Character
        local r = char2 and char2:FindFirstChild("HumanoidRootPart")
        if not r then return end
        local cam = workspace.CurrentCamera
        local cf  = cam.CFrame
        local spd = State.FlySpeed
        local hum2 = char2:FindFirstChildOfClass("Humanoid")
        local md = hum2 and hum2.MoveDirection or Vector3.zero

        local flatLook = Vector3.new(cf.LookVector.X, 0, cf.LookVector.Z)
        flatLook = flatLook.Magnitude > 0 and flatLook.Unit or Vector3.zero
        local flatRight = Vector3.new(cf.RightVector.X, 0, cf.RightVector.Z)
        flatRight = flatRight.Magnitude > 0 and flatRight.Unit or Vector3.zero

        local fwd    = md:Dot(flatLook)
        local strafe = md:Dot(flatRight)
        local move   = cf.LookVector * fwd + cf.RightVector * strafe

        if move.Magnitude > 0 then
            bv.Velocity = move.Unit * spd
        else
            bv.Velocity = Vector3.zero
        end
        bg.CFrame = cf - cf.Position + r.Position
    end)
end

local flingSteppedConn = nil
local function setFlingGhost(on)
    -- ponytail: trik FE Fling klasik — collision sendiri dimatikan biar kita
    -- nembus ke tubuh target tanpa pantulan solver (sumber "geter-geter")
    if flingSteppedConn then flingSteppedConn:Disconnect() flingSteppedConn = nil end
    if not on then return end
    flingSteppedConn = RunService.Stepped:Connect(function()
        local c = player.Character
        if not c then return end
        for _, p in ipairs(c:GetDescendants()) do
            if p:IsA("BasePart") then p.CanCollide = false end
        end
    end)
end

local function flingLoop()
    -- ponytail: resep JackMcJagger15 — collision sendiri off + BodyThrust
    -- konstan di HRP; velocity acak malah dilawan solver dan nggak ngefek
    local char   = player.Character
    local myRoot = char and char:FindFirstChild("HumanoidRootPart")
    if not myRoot then return end
    setFlingGhost(true)
    local thrust = Instance.new("BodyThrust")
    thrust.Force   = Vector3.new(State.FlingPower, 0, State.FlingPower)
    thrust.Location = Vector3.new(0, 0, -1)
    thrust.Parent   = myRoot
    while State.FlingOn do
        local c     = player.Character
        local r     = c and c:FindFirstChild("HumanoidRootPart")
        local tChar = State.FlingTarget and State.FlingTarget.Character
        local tRoot = tChar and tChar:FindFirstChild("HumanoidRootPart")
        if r and tRoot then
            if thrust.Parent ~= r then thrust.Parent = r end
            r.CFrame = tRoot.CFrame * CFrame.new(0, 0, -1)
        end
        task.wait()
    end
    pcall(function() thrust:Destroy() end)
    setFlingGhost(false)
end

local hitboxConn = nil
local hitboxOrig = {}
local function restoreHitboxes()
    for hrp, o in pairs(hitboxOrig) do
        if hrp.Parent then
            hrp.Size         = o.size
            hrp.Transparency = o.transparency
            hrp.Material     = o.material
            hrp.Massless     = o.massless
            hrp.CanCollide   = o.canCollide
        end
    end
    hitboxOrig = {}
end

local function setHitboxes(on)
    State.Hitbox = on
    if hitboxConn then hitboxConn:Disconnect() hitboxConn = nil end
    if on then
        hitboxConn = RunService.RenderStepped:Connect(function()
            for _, other in ipairs(Players:GetPlayers()) do
                if other ~= player then
                    local char2 = other.Character
                    local hrp = char2 and char2:FindFirstChild("HumanoidRootPart")
                    if hrp then
                        hitboxOrig[hrp] = hitboxOrig[hrp] or {
                            size = hrp.Size, transparency = hrp.Transparency,
                            material = hrp.Material, massless = hrp.Massless,
                            canCollide = hrp.CanCollide,
                        }
                        local s = State.HitboxSize
                        hrp.Size         = Vector3.new(s, s, s)
                        hrp.Transparency = 0.75
                        hrp.Massless     = true
                        hrp.CanCollide   = false
                    end
                end
            end
        end)
    else
        restoreHitboxes()
    end
end

local KORBLOX_MESH = {
    RightUpperLeg = { mesh = "rbxassetid://902942096", tex = "rbxassetid://902843398" },
    RightLowerLeg = { mesh = "rbxassetid://902942093", hide = true },
    RightFoot     = { mesh = "rbxassetid://902942089", hide = true },
}
local korbloxOriginals = {}
local cosmeticSpawnConn = nil

local function applyHeadlessChar(char)
    local head = char:FindFirstChild("Head") or char:WaitForChild("Head", 5)
    if not head then return end
    head.Transparency = 1
    for _, v in ipairs(head:GetChildren()) do
        if v:IsA("Decal") then v.Transparency = 1 end
    end
end

local function clearHeadlessChar(char)
    local head = char:FindFirstChild("Head")
    if not head then return end
    head.Transparency = 0
    for _, v in ipairs(head:GetChildren()) do
        if v:IsA("Decal") then v.Transparency = 0 end
    end
end

local function applyKorbloxChar(char)
    for name, cfg in pairs(KORBLOX_MESH) do
        local part = char:FindFirstChild(name) or char:WaitForChild(name, 5)
        if part then
            if not korbloxOriginals[part] then
                local okM, origMesh = pcall(function() return part.MeshId end)
                local okT, origTex = pcall(function() return part.TextureID end)
                korbloxOriginals[part] = {
                    mesh   = okM and origMesh or nil,
                    tex    = okT and origTex or nil,
                    transp = part.Transparency,
                }
            end
            pcall(function()
                part.MeshId = cfg.mesh
                if cfg.tex then part.TextureID = cfg.tex end
            end)
            if cfg.hide then part.Transparency = 1 end
        end
    end
end

local function clearKorbloxChar(char)
    for part, orig in pairs(korbloxOriginals) do
        if part.Parent then
            pcall(function()
                if orig.mesh then part.MeshId = orig.mesh end
                if orig.tex then part.TextureID = orig.tex end
            end)
            part.Transparency = orig.transp
        end
    end
    korbloxOriginals = {}
end

local function refreshCosmeticSpawn()
    if cosmeticSpawnConn then cosmeticSpawnConn:Disconnect() cosmeticSpawnConn = nil end
    if State.Headless or State.Korblox then
        cosmeticSpawnConn = player.CharacterAdded:Connect(function(c)
            task.wait(0.5)
            if State.Headless then applyHeadlessChar(c) end
            if State.Korblox then applyKorbloxChar(c) end
        end)
    end
end

local function setHeadless(on)
    State.Headless = on
    local char = player.Character
    if char then
        if on then applyHeadlessChar(char) else clearHeadlessChar(char) end
    end
    refreshCosmeticSpawn()
end

local function setKorblox(on)
    State.Korblox = on
    local char = player.Character
    if char then
        if on then applyKorbloxChar(char) else clearKorbloxChar(char) end
    end
    refreshCosmeticSpawn()
end

local ESP_NAME    = "danar_esp_highlight"
local ESP_BB_NAME = "danar_esp_bb"

local function attachESP(char)
    if not State.ESP then return end
    if char:FindFirstChild(ESP_NAME) then return end

    local root2 = char:FindFirstChild("HumanoidRootPart")
    local head2 = char:FindFirstChild("Head")
    if not root2 then return end

    local plrName = ""
    for _, p in ipairs(Players:GetPlayers()) do
        if p.Character == char then plrName = p.Name break end
    end

    local hl = Instance.new("Highlight")
    hl.Name             = ESP_NAME
    hl.FillColor        = THEME.Accent
    hl.FillTransparency = 0.65
    hl.OutlineColor     = Color3.new(1, 1, 1)
    hl.DepthMode        = Enum.HighlightDepthMode.AlwaysOnTop
    hl.Adornee          = char
    hl.Parent           = char

    local adornTarget = head2 or root2

    local bb = Instance.new("BillboardGui")
    bb.Name            = ESP_BB_NAME
    bb.AlwaysOnTop     = true
    bb.MaxDistance     = 600
    bb.Size            = UDim2.new(0, 120, 0, 22)
    bb.StudsOffsetWorldSpace = Vector3.new(0, 0, 0)
    bb.StudsOffset     = Vector3.new(0, 3.2, 0)
    bb.Adornee         = adornTarget
    bb.Parent          = char

    local nameLabel = Instance.new("TextLabel")
    nameLabel.BackgroundTransparency = 1
    nameLabel.Size            = UDim2.new(1, 0, 1, 0)
    nameLabel.Font            = Enum.Font.GothamBold
    nameLabel.TextSize        = 13
    nameLabel.TextColor3      = Color3.fromRGB(255, 255, 255)
    nameLabel.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
    nameLabel.TextStrokeTransparency = 0.4
    nameLabel.Text            = plrName
    nameLabel.TextScaled      = false
    nameLabel.Parent          = bb

    local distConn
    distConn = RunService.Heartbeat:Connect(function()
        if not bb.Parent or not State.ESP then
            distConn:Disconnect()
            return
        end
        local cam = workspace.CurrentCamera
        local myRoot = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
        if not myRoot or not adornTarget.Parent then return end

        local dist = (adornTarget.Position - myRoot.Position).Magnitude
        local scale = math.clamp(1 - (dist / 600) * 0.6, 0.4, 1)
        local sz    = math.floor(13 * scale + 0.5)
        if nameLabel.TextSize ~= sz then
            nameLabel.TextSize = sz
        end
        local alpha = math.clamp(1 - (dist / 600), 0.25, 1)
        nameLabel.TextTransparency = 1 - alpha
    end)
end

        if inst.Name == ESP_NAME or inst.Name == ESP_BB_NAME then
            inst:Destroy()
        end
    end
end

local function setFPSBoost(on)
    if on then matOrig = {} end
    for _, d in ipairs(workspace:GetDescendants()) do
        if d:IsA("ParticleEmitter") or d:IsA("Trail") or d:IsA("Smoke")
           or d:IsA("Fire") or d:IsA("Sparkles") then
            d.Enabled = not on
        elseif d:IsA("BasePart") then
            if on then
                matOrig[d] = { material = d.Material, reflectance = d.Reflectance }
                d.Material = Enum.Material.SmoothPlastic
                d.Reflectance = 0
            else
                local orig = matOrig[d]
                if orig then d.Material = orig.material d.Reflectance = orig.reflectance end
            end
        end
    end
    if not on then matOrig = {} end
end

local stickyTarget = nil

local function measureTarget(p)
    local cam = workspace.CurrentCamera
    if not cam then return nil end
    local char = p.Character
    if not char then return nil end
    local head = char:FindFirstChild("Head")
    local hum  = char:FindFirstChildOfClass("Humanoid")
    if not head or not hum or hum.Health <= 0 then return nil end
    if State.AimTeamCheck and p.Team == player.Team then return nil end
    if State.AimWallCheck then
        local rparams = RaycastParams.new()
        rparams.FilterType = Enum.RaycastFilterType.Exclude
        rparams.FilterDescendantsInstances = { player.Character, char }
        local ray = workspace:Raycast(cam.CFrame.Position,
            head.Position - cam.CFrame.Position, rparams)
        if ray then return nil end
    end
    local screenPos, onScreen = cam:WorldToViewportPoint(head.Position)
    if not onScreen then return nil end
    local vpCenter = Vector2.new(cam.ViewportSize.X * 0.5, cam.ViewportSize.Y * 0.5)
    local dist2D = (Vector2.new(screenPos.X, screenPos.Y) - vpCenter).Magnitude
    if dist2D >= State.AimFOV then return nil end
    return dist2D
end

local function getNearestTarget()
    if stickyTarget and stickyTarget.Parent then
        if measureTarget(stickyTarget) then return stickyTarget end
    end
    stickyTarget = nil
    local cam = workspace.CurrentCamera
    local myRoot = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
    if not cam or not myRoot then return nil end
    local best, bestDist = nil, State.AimFOV
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= player then
            local d = measureTarget(p)
            if d and d < bestDist then
                bestDist = d
                best = p
            end
        end
    end
    stickyTarget = best
    return best
end

local function getMM2Role(plr)
    local char = plr.Character
    if not char then return "innocent" end
    for _, tool in ipairs(char:GetChildren()) do
        if tool:IsA("Tool") then
            local n = tool.Name:lower()
            if n:find("knife") or n:find("blade") or n:find("murder") then
                return "murderer"
            end
            if n:find("gun") or n:find("sheriff") or n:find("revolver") then
                return "sheriff"
            end
        end
    end
    local bp = plr:FindFirstChild("Backpack")
    if bp then
        for _, tool in ipairs(bp:GetChildren()) do
            if tool:IsA("Tool") then
                local n = tool.Name:lower()
                if n:find("knife") or n:find("blade") or n:find("murder") then
                    return "murderer"
                end
                if n:find("gun") or n:find("sheriff") or n:find("revolver") then
                    return "sheriff"
                end
            end
        end
    end
    return "innocent"
end

local function autoGrabGunLoop()
    while MM2.AutoGrabGun do
        local gun = findDroppedGun()
        if gun then
            local handle = gun:FindFirstChild("Handle")
            local char   = player.Character
            local root   = char and char:FindFirstChild("HumanoidRootPart")
            if handle and root then
                local dist = (handle.Position - root.Position).Magnitude
                if dist > 3 then
                    -- ponytail: durasi proporsional jarak, sama kayak coin farm
                    local sps = math.clamp(MM2.GrabSpeed, 1, 20) * 7
                    local tw = TweenService:Create(root,
                        TweenInfo.new(math.max(dist / sps, 0.08), Enum.EasingStyle.Linear),
                        { CFrame = CFrame.new(handle.Position + Vector3.new(0, 3, 0)) })
                    tw:Play()
                    tw.Completed:Wait()
                end
                task.wait(0.05)
                local hum = char and char:FindFirstChildOfClass("Humanoid")
                if hum then hum:EquipTool(gun) end
            end
        end
        task.wait(0.4)
    end
end

                                        or game:GetService("ReplicatedStorage"):FindFirstChild("Shoot", true)

local function forceTouch(root, coin)
    if firetouchinterest then
        pcall(firetouchinterest, root, coin, 0)
        pcall(firetouchinterest, root, coin, 1)
    end
    local cd = coin:FindFirstChildOfClass("ClickDetector")
    if cd and fireclickdetector then pcall(fireclickdetector, cd) end
    local pp = coin:FindFirstChildOfClass("ProximityPrompt")
        or (coin.Parent and coin.Parent:FindFirstChildOfClass("ProximityPrompt"))
    if pp and fireproximityprompt then pcall(fireproximityprompt, pp) end
end

local function coinFarmLoop()
    ensureSafePart()
    local safeCF = safePart.CFrame + Vector3.new(0, 3, 0)
    while MM2.CoinFarm do
        local cc = roundActive and not bagFull and findCoinContainer()
        if cc then
            pcTp(safeCF)
            for _, coin in ipairs(cc:GetChildren()) do
                if not MM2.CoinFarm or bagFull then break end
                if coin:GetAttribute("CoinID") == MM2.CoinType
                   and coin:FindFirstChild("TouchInterest") then
                    for _ = 1, 7 do
                        if not MM2.CoinFarm then break end
                        pcTp(coin.CFrame + Vector3.new(0, 1, 0))
                        task.wait(0.03)
                    end
                    local char = player.Character
                    local root = char and char:FindFirstChild("HumanoidRootPart")
                    if root then forceTouch(root, coin) end
                    break -- satu coin per sweep, pola NoCapital
                end
            end
            pcTp(safeCF)
            task.wait(math.max(MM2.CoinDelay, 1) * (0.9 + math.random() * 0.2))
        else
            -- nunggu ronde / tas dipulihkan: parkir aman di bawah map
            if player.Character then pcTp(safeCF) end
            if bagFull and MM2.BagFullReset then
                local hum = player.Character and player.Character:FindFirstChildOfClass("Humanoid")
                if hum then hum.Health = 0 end
            end
            task.wait(0.5)
        end
    end
end

        local cacheName = "danarhub_bg_" .. string.format("%x", h) .. ".png"

        or i.UserInputType == Enum.UserInputType.Touch then sliding = false end
    end))

    stroke(kbBtn, THEME.Accent2, 1)

        if not listening then return end

        listening = false

            current = input.KeyCode

        end
        kbBtn.Text = typeof(current) == "EnumItem" and current.Name or "None"
    end))

    kbBtn.Activated:Connect(function()
        listening = true
        kbBtn.Text = "..."
    end)

local uiOpenState = true

local function setOpen(open)
    if animBusy or open == uiOpenState or not screenGui.Parent then return end
    uiOpenState = open
    animBusy    = true
    if open then
        local ptw = TweenService:Create(pill,
            TweenInfo.new(0.12, Enum.EasingStyle.Quad, Enum.EasingDirection.In),
            { Size = UDim2.fromOffset(PILL_W, 0) })
        ptw.Completed:Connect(function() pill.Visible = false end)
        ptw:Play()
        main.Visible    = true
        main.AnchorPoint = Vector2.new(0.5, 0.5)
        main.Position   = savedPos or UDim2.new(0.5, 0, 0.5, 0)
        main.Size       = UDim2.fromOffset(MAIN_W, 0)
        local tw = TweenService:Create(main,
            TweenInfo.new(0.3, Enum.EasingStyle.Back, Enum.EasingDirection.Out),
            { Size = UDim2.fromOffset(MAIN_W, MAIN_H) })
        tw.Completed:Connect(function() animBusy = false end)
        tw:Play()
    else
        pill.Visible = true
        pill.Size    = UDim2.fromOffset(PILL_W, 0)
        TweenService:Create(pill,
            TweenInfo.new(0.24, Enum.EasingStyle.Back, Enum.EasingDirection.Out),
            { Size = UDim2.fromOffset(PILL_W, PILL_H) }):Play()
        local tw = TweenService:Create(main,
            TweenInfo.new(0.18, Enum.EasingStyle.Quad, Enum.EasingDirection.In),
            { Size = UDim2.fromOffset(MAIN_W, 0) })
        tw.Completed:Connect(function()
            main.Visible = false
            main.Size    = UDim2.fromOffset(MAIN_W, MAIN_H)
            animBusy     = false
        end)
        tw:Play()
    end
end

local function runCleanup()
    State.FlingOn = false
    if BF then
        BF.FarmOn = false
        BF.AttackOn = false
        BF.FastAtk = false
        BF.HitboxExpand = false
        if BF._restoreWrap then pcall(BF._restoreWrap) end
    end
    do
        local c  = player.Character
        local r  = c and c:FindFirstChild("HumanoidRootPart")
        local th = r and r:FindFirstChildOfClass("BodyThrust")
        if th then pcall(function() th:Destroy() end) end
        if flingSteppedConn then flingSteppedConn:Disconnect() flingSteppedConn = nil end
    end
    State.Noclip = false
    State.InfiniteJump = false
    State.ESP = false
    if State.Fly then State.Fly = false applyFly(false) end
    if State.TPWalk then setTPWalk(false) end
    if State.AntiFling then setAntiFling(false) end
    if MM2.AimLockMurder then setAimLockMurder(false) end
    if State.Headless then setHeadless(false) end
    if State.Korblox then setKorblox(false) end
    if State.Fullbright then State.Fullbright = false setFullbright(false) end
    if State.FPSBoost then State.FPSBoost = false setFPSBoost(false) end
    if State.Hitbox then State.Hitbox = false setHitboxes(false) end
    clearESP()
    applySpeed()
    applyJump()
    if getgenv().DANAR_SET_ANTIKICK then getgenv().DANAR_SET_ANTIKICK(false) end
    State.AimOn = false
    disableFOVCircle()
    MM2.RoleESP       = false
    if roleESPJoinConn then
        roleESPJoinConn:Disconnect()
        roleESPJoinConn = nil
    end
    for _, conn in pairs(roleESPConns) do conn:Disconnect() end
    roleESPConns = {}
    MM2.AutoGrabGun   = false
    MM2.GunESP        = false
    clearGunESP()
    if gunTrackerLine then
        gunTrackerLine:Remove()
        gunTrackerLine = nil
    end
    MM2.ShootMurderer = false
    MM2.CoinFarm      = false
    pcall(function() RunService:Set3dRenderingEnabled(true) end)
    if safePart then pcall(function() safePart:Destroy() end) safePart = nil end
    clearRoleESP()
    clearGunESP()
    for _, conn in pairs(playerHooks) do pcall(function() conn:Disconnect() end) end
    table.clear(playerHooks)
    if espPlayerConn then espPlayerConn:Disconnect() espPlayerConn = nil end
    if espRemoveConn then espRemoveConn:Disconnect() espRemoveConn = nil end
    for _, conn in pairs(espCharConns) do pcall(function() conn:Disconnect() end) end
    table.clear(espCharConns)
    for _, conn in ipairs(connections) do
        pcall(function() conn:Disconnect() end)
    end
    table.clear(connections)
end

screenGui.Destroying:Connect(runCleanup)
